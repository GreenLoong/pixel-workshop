# 保留可执行文件原有位置，部署脚本和开发者命令无需更改路径。
get_property(check_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(check IN LISTS check_targets)
    set_target_properties(${check} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
    if(NOT check STREQUAL "package_checks")
        set_target_properties(${check} PROPERTIES FOLDER "Tests")
    endif()
    if(MSVC)
        target_compile_options(${check} PRIVATE /utf-8)
    endif()
    get_target_property(check_sources ${check} SOURCES)
    source_group(TREE "${PROJECT_SOURCE_DIR}" FILES ${check_sources})
    if(IMAGEBATCHTOOL_BUILD_TESTS)
        add_test(NAME ${check} COMMAND ${check})
    endif()
endforeach()
